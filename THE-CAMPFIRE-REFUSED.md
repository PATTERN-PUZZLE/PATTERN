  <script>
    let currentOutput = '';

    function getMode() {
      const radios = document.getElementsByName('mode');
      for (const r of radios) { if (r.checked) return r.value; }
      return 'fire';
    }

    function toggleCrystalInput() {
      const wrap = document.getElementById('crystalInputWrap');
      wrap.style.display = getMode() === 'crystal' ? 'block' : 'none';
    }

    document.querySelectorAll('input[name="mode"]').forEach(el => {
      el.addEventListener('change', toggleCrystalInput);
    });
    toggleCrystalInput();

    function makeId() {
      return 'crystal-' + Date.now().toString(36);
    }

    function makeTag() {
      return '◆' + Math.floor(1000 + Math.random() * 9000);
    }

    function makeLink(id, content) {
      const encoded = btoa(encodeURIComponent(content)).replace(/=+$/, '');
      return window.location.origin + window.location.pathname + '#c=' + id + '&d=' + encoded;
    }

    function createBlock() {
      const raw = document.getElementById('raw').value.trim();
      if (!raw) { setStatus('🪵💛 Paste raw text first.'); return; }

      const mode = getMode();
      const id = makeId();
      const tag = makeTag();
      const date = new Date().toISOString().split('T')[0];

      let crystal = '';
      if (mode === 'crystal') {
        crystal = document.getElementById('crystalLine').value.trim();
        if (!crystal) { setStatus('💎 Write your crystal line first.'); return; }
      }

      const snippet = raw.length > 120 ? raw.slice(0, 120) + '...' : raw;
      const modeLabel = mode === 'fire' ? '🔥 FIRE KEEP' : '💎 CRYSTAL PRESS';
      const essence = crystal ? crystal : '(raw voice — pass as is)';

      let output = '```\n';
      output += 'SEED-CRYSTAL BLOCK — ' + date + '\n';
      output += 'ID: ' + id + '\n';
      output += 'MODE: ' + modeLabel + '\n';
      output += 'BOTTOM TAG: ' + tag + '\n\n';
      output += 'RAW SNIPPET: ' + snippet + '\n\n';
      output += 'CRYSTAL: ' + essence + '\n\n';
      if (mode === 'fire') {
        output += 'RAW TEXT:\n' + raw + '\n\n';
      }
      output += 'SUGGESTED HOME: Bucket / Extra Patterns / Origin Fire / Foolish Wisdom\n';
      output += 'ONE-LINE RELATION: (add later)\n\n';
      output += 'SHARE: ' + makeLink(id, raw + '\n\nCRYSTAL: ' + essence) + '\n';
      output += '```';

      currentOutput = output;
      document.getElementById('output').textContent = output;
      setStatus('✅ Block created. Use 📥 Copy Output to save or share.');
    }

    function preparePrompt() {
      const raw = document.getElementById('raw').value.trim();
      if (!raw) { setStatus('🪵💛 Paste raw text first.'); return; }

      const prompt = 'Distill this. Keep the heat. Preserve the voice. Return:\n1. A one-sentence crystal\n2. A one-line placement note\n3. A bottom tag\n\nRaw:\n' + raw;

      currentOutput = prompt;
      document.getElementById('output').textContent = prompt;
      setStatus('📋 Prompt ready. Copy it and paste to your calibrated instance.');
    }

    function copyOutput() {
      if (!currentOutput) { setStatus('Nothing to copy yet.'); return; }
      navigator.clipboard.writeText(currentOutput).then(() => {
        setStatus('📥 Copied to clipboard.');
      }).catch(() => {
        setStatus('⚠️ Copy failed. Select the output manually.');
      });
    }

    function setStatus(msg) {
      document.getElementById('status').textContent = msg;
    }

    // Load from shared link if present
    (function loadFromHash() {
      const hash = window.location.hash;
      if (!hash || !hash.includes('#c=')) return;
      const params = new URLSearchParams(hash.slice(1));
      const id = params.get('c');
      const data = params.get('d');
      if (data) {
        try {
          const decoded = decodeURIComponent(atob(data.replace(/-/g, '+').replace(/_/g, '/')));
          document.getElementById('output').textContent = 'SHARED CRYSTAL #' + id + '\n\n' + decoded;
          setStatus('🔦 Shared crystal received.');
        } catch (e) {
          setStatus('⚠️ Could not decode shared crystal.');
        }
      }
    })();
